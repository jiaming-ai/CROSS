cmake_minimum_required(VERSION 3.10)
project(cross_baselines)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_BUILD_TYPE Release)
set(ORB_DIR ${CMAKE_SOURCE_DIR}/../../third_party/ORB_SLAM3)
list(APPEND CMAKE_PREFIX_PATH ${CMAKE_SOURCE_DIR}/../../third_party/install)
find_package(OpenCV 4 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Pangolin REQUIRED)
include_directories(${ORB_DIR} ${ORB_DIR}/include ${ORB_DIR}/include/CameraModels ${ORB_DIR}/Thirdparty/Sophus ${EIGEN3_INCLUDE_DIR} ${Pangolin_INCLUDE_DIRS})
add_executable(orbslam3_reloc orbslam3_reloc.cc)
target_link_libraries(orbslam3_reloc ${ORB_DIR}/lib/libORB_SLAM3.so ${ORB_DIR}/Thirdparty/DBoW2/lib/libDBoW2.so ${ORB_DIR}/Thirdparty/g2o/lib/libg2o.so ${OpenCV_LIBS} ${Pangolin_LIBRARIES} -lboost_serialization -lcrypto)
find_package(RTABMap QUIET)
if(RTABMap_FOUND)
  add_executable(rtabmap_reloc rtabmap_reloc.cc)
  target_link_libraries(rtabmap_reloc rtabmap::core ${OpenCV_LIBS})
else()
  message(WARNING "RTABMap not found; rtabmap_reloc not built")
endif()
